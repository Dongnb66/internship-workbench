-- 每张表同一套三件套，10 张表 × 1 条 ALL 策略
ALTER TABLE public.messages
  ADD COLUMN IF NOT EXISTS owner_id TEXT NOT NULL
  DEFAULT auth.uid();

GRANT SELECT, INSERT, UPDATE, DELETE
  ON public.messages TO authenticated, anon;

DROP POLICY IF EXISTS messages_own ON public.messages;
CREATE POLICY messages_own ON public.messages
  FOR ALL
  USING (owner_id = auth.uid())
  WITH CHECK (owner_id = auth.uid());
USING 管「能读哪些行」，WITH CHECK 管「能写成哪些行」。
只写 USING 的话，仍可 INSERT 一条归属别人的记录。两段都必须写。
▼ 业务流程

⑥ 核心业务流程

一条线管到底
STEP 1 岗位池 粘贴 JD → 本地规则打预筛分 → 勾选批量送 AI 七维深评 → 匹配度 ≥ 55 的转入投递
STEP 2 投递看板 五列看板拖拽改阶段 · 复制打招呼话术 → 过 6 条自检清单 → 登记发送
STEP 3 沟通台账 会话时间线记录每次往来 · 阶段与下一步日期自动联动 · 节奏卡控发送频率
STEP 4 面试跟进 日程 / 轮次 / 结果登记 · AI 押题 · 当晚写复盘由 AI 整理成结构化清单
STEP 5 Offer 对比 七维评分排序 → 看清自己在乎成长、钱还是通勤
横切关注点：超期未回复（≥7 天）在看板卡片标红并在总览集中提醒；提醒日历汇总岗位截止、面试笔试、跟进与自建待办。
